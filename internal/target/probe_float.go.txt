
func PROBECompareFloat[T ~float32 | ~float64](site int, mask uint64, left, right T) uint64 {
	var order uint64 = 32
	if left < right {
		order = 1
	} else if left > right {
		order = 4
	} else if left == right {
		order = 2
	}
	state := PROBEGetState()
	if state.slots != nil && site >= 0 && site*5+5 <= len(state.slots) {
		if state.site < 0 {
			PROBERecord(&state.slots[site*5], order)
		} else if state.site == site {
			return PROBEActivate(state, site, mask, order, order)
		}
	}
	return order & mask
}
