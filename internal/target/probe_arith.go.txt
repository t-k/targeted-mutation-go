
type PROBEUint = uint

func PROBEArith[T PROBEInteger](site int, origOp uint64, left, right T) T {
	state := PROBEGetState()
	if state.slots == nil || site < 0 || site*5+5 > len(state.slots) {
		return PROBEApplyArith(origOp, left, right)
	}
	if state.site >= 0 {
		if state.site != site {
			return PROBEApplyArith(origOp, left, right)
		}
		origPanics := (origOp == 4 || origOp == 5) && right == 0
		mutPanics := (state.mask == 4 || state.mask == 5) && right == 0
		if origPanics != mutPanics {
			PROBERecord(&state.slots[site*5], 24)
			return PROBEApplyArith(state.mask, left, right)
		}
		if origPanics && mutPanics {
			PROBERecord(&state.slots[site*5], 8)
			return PROBEApplyArith(state.mask, left, right)
		}
		orig := PROBEApplyArith(origOp, left, right)
		mut := PROBEApplyArith(state.mask, left, right)
		flags := uint64(8)
		if mut != orig {
			flags |= 16
		}
		PROBERecord(&state.slots[site*5], flags)
		return mut
	}
	slot := state.slots[site*5 : site*5+5]
	PROBERecord(&slot[0], 1)
	if origOp <= 5 {
		const arithAll = (uint64(1) << 1) | (1 << 2) | (1 << 3) | (1 << 4) | (1 << 5)
		fullMask := arithAll &^ (uint64(1) << origOp)
		if (origOp == 4 || origOp == 5) && right == 0 {
			PROBERecord(&slot[1], (1<<1)|(1<<2)|(1<<3))
			return PROBEApplyArith(origOp, left, right)
		}
		orig := PROBEApplyArith(origOp, left, right)
		if PROBEatomic.LoadUint64(&slot[1])&fullMask != fullMask {
			var diff uint64
			if left+right != orig {
				diff |= 1 << 1
			}
			if left-right != orig {
				diff |= 1 << 2
			}
			if left*right != orig {
				diff |= 1 << 3
			}
			if right == 0 {
				diff |= (1 << 4) | (1 << 5)
			} else {
				if left/right != orig {
					diff |= 1 << 4
				}
				if left%right != orig {
					diff |= 1 << 5
				}
			}
			if diff != 0 {
				PROBERecord(&slot[1], diff)
			}
		}
		return orig
	}
	const bitAll = (uint64(1) << 6) | (1 << 7) | (1 << 8) | (1 << 9)
	fullMask := bitAll &^ (uint64(1) << origOp)
	orig := PROBEApplyArith(origOp, left, right)
	if PROBEatomic.LoadUint64(&slot[1])&fullMask != fullMask {
		var diff uint64
		if left&right != orig {
			diff |= 1 << 6
		}
		if left|right != orig {
			diff |= 1 << 7
		}
		if left^right != orig {
			diff |= 1 << 8
		}
		if left&^right != orig {
			diff |= 1 << 9
		}
		if diff != 0 {
			PROBERecord(&slot[1], diff)
		}
	}
	return orig
}

func PROBEShift[T, U PROBEInteger](site int, origOp uint64, left T, right U) T {
	state := PROBEGetState()
	if state.slots == nil || site < 0 || site*5+5 > len(state.slots) {
		if origOp == 10 {
			return left << right
		}
		return left >> right
	}
	if state.site >= 0 {
		if state.site != site {
			if origOp == 10 {
				return left << right
			}
			return left >> right
		}
		if right < 0 {
			PROBERecord(&state.slots[site*5], 8)
			if state.mask == 10 {
				return left << right
			}
			return left >> right
		}
		var orig, mut T
		if origOp == 10 {
			orig = left << right
		} else {
			orig = left >> right
		}
		if state.mask == 10 {
			mut = left << right
		} else {
			mut = left >> right
		}
		flags := uint64(8)
		if mut != orig {
			flags |= 16
		}
		PROBERecord(&state.slots[site*5], flags)
		return mut
	}
	slot := state.slots[site*5 : site*5+5]
	PROBERecord(&slot[0], 1)
	if right < 0 {
		if origOp == 10 {
			return left << right
		}
		return left >> right
	}
	otherOp := uint64(11)
	var orig, other T
	if origOp == 10 {
		orig = left << right
		other = left >> right
	} else {
		otherOp = 10
		orig = left >> right
		other = left << right
	}
	if other != orig {
		PROBERecord(&slot[1], uint64(1)<<otherOp)
	}
	return orig
}

func PROBEAssign[T PROBEInteger](site int, origOp uint64, dst *T, right T) {
	left := *dst
	state := PROBEGetState()
	if state.slots == nil || site < 0 || site*5+5 > len(state.slots) {
		*dst = PROBEApplyArith(origOp, left, right)
		return
	}
	if state.site >= 0 {
		if state.site != site {
			*dst = PROBEApplyArith(origOp, left, right)
			return
		}
		origPanics := (origOp == 4 || origOp == 5) && right == 0
		mutPanics := (state.mask == 4 || state.mask == 5) && right == 0
		if origPanics != mutPanics {
			PROBERecord(&state.slots[site*5], 24)
			*dst = PROBEApplyArith(state.mask, left, right)
			return
		}
		if origPanics && mutPanics {
			PROBERecord(&state.slots[site*5], 8)
			*dst = PROBEApplyArith(state.mask, left, right)
			return
		}
		orig := PROBEApplyArith(origOp, left, right)
		mut := PROBEApplyArith(state.mask, left, right)
		flags := uint64(8)
		if mut != orig {
			flags |= 16
		}
		PROBERecord(&state.slots[site*5], flags)
		*dst = mut
		return
	}
	slot := state.slots[site*5 : site*5+5]
	PROBERecord(&slot[0], 1)
	if origOp <= 5 {
		const arithAll = (uint64(1) << 1) | (1 << 2) | (1 << 3) | (1 << 4) | (1 << 5) | (1 << 10)
		fullMask := arithAll &^ (uint64(1) << origOp)
		if (origOp == 4 || origOp == 5) && right == 0 {
			PROBERecord(&slot[1], (1<<1)|(1<<2)|(1<<3)|(1<<10))
			*dst = PROBEApplyArith(origOp, left, right)
			return
		}
		orig := PROBEApplyArith(origOp, left, right)
		if PROBEatomic.LoadUint64(&slot[1])&fullMask != fullMask {
			var diff uint64
			if left+right != orig {
				diff |= 1 << 1
			}
			if left-right != orig {
				diff |= 1 << 2
			}
			if left*right != orig {
				diff |= 1 << 3
			}
			if right == 0 {
				diff |= (1 << 4) | (1 << 5)
			} else {
				if left/right != orig {
					diff |= 1 << 4
				}
				if left%right != orig {
					diff |= 1 << 5
				}
			}
			if right != orig {
				diff |= 1 << 10
			}
			if diff != 0 {
				PROBERecord(&slot[1], diff)
			}
		}
		*dst = orig
		return
	}
	const bitAll = (uint64(1) << 6) | (1 << 7) | (1 << 8) | (1 << 9) | (1 << 10)
	fullMask := bitAll &^ (uint64(1) << origOp)
	orig := PROBEApplyArith(origOp, left, right)
	if PROBEatomic.LoadUint64(&slot[1])&fullMask != fullMask {
		var diff uint64
		if left&right != orig {
			diff |= 1 << 6
		}
		if left|right != orig {
			diff |= 1 << 7
		}
		if left^right != orig {
			diff |= 1 << 8
		}
		if left&^right != orig {
			diff |= 1 << 9
		}
		if right != orig {
			diff |= 1 << 10
		}
		if diff != 0 {
			PROBERecord(&slot[1], diff)
		}
	}
	*dst = orig
}

func PROBEApplyArith[T PROBEInteger](op uint64, left, right T) T {
	switch op {
	case 1:
		return left + right
	case 2:
		return left - right
	case 3:
		return left * right
	case 4:
		return left / right
	case 5:
		return left % right
	case 6:
		return left & right
	case 7:
		return left | right
	case 8:
		return left ^ right
	case 9:
		return left &^ right
	case 10:
		return right
	}
	return left
}
