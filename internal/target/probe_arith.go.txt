
func PROBEArith[T PROBEInteger](site int, origOp uint64, left, right T) T {
	orig := PROBEApplyArith(origOp, left, right)
	state := PROBEGetState()
	if state.slots == nil || site < 0 || site*5+5 > len(state.slots) {
		return orig
	}
	if state.site >= 0 {
		if state.site != site {
			return orig
		}
		if (state.mask == 4 || state.mask == 5) && right == 0 {
			PROBERecord(&state.slots[site*5], 24)
			return PROBEApplyArith(state.mask, left, right)
		}
		mut := PROBEApplyArith(state.mask, left, right)
		flags := uint64(8)
		if mut != orig {
			flags |= 16
		}
		PROBERecord(&state.slots[site*5], flags)
		return mut
	}
	PROBERecord(&state.slots[site*5], 1)
	return orig
}

func PROBEApplyArith[T PROBEInteger](op uint64, left, right T) T {
	switch op {
	case 1:
		return left + right
	case 2:
		return left - right
	case 3:
		return left * right
	case 4:
		return left / right
	case 5:
		return left % right
	case 6:
		return left & right
	case 7:
		return left | right
	case 8:
		return left ^ right
	case 9:
		return left &^ right
	}
	return left
}
