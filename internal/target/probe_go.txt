package PROBE_PACKAGE

import (
	PROBEos "os"
	PROBEruntime "runtime"
	PROBEstrconv "strconv"
	PROBEstrings "strings"
	PROBEatomic "sync/atomic"
	PROBEsyscall "syscall"
	PROBEunsafe "unsafe"
)

type PROBEInteger interface {
	~int | ~int8 | ~int16 | ~int32 | ~int64 | ~uint | ~uint8 | ~uint16 | ~uint32 | ~uint64 | ~uintptr
}

type PROBEState struct {
	slots []uint64
	site  int
	mask  uint64
	pivot uint64
	mode  uint8
}

var PROBEStateOnce uint32
var PROBEStateValue PROBEState

func PROBEGetState() *PROBEState {
	if PROBEatomic.LoadUint32(&PROBEStateOnce) == 2 {
		return &PROBEStateValue
	}
	if PROBEatomic.CompareAndSwapUint32(&PROBEStateOnce, 0, 1) {
		PROBEStateValue = PROBEOpen()
		PROBEatomic.StoreUint32(&PROBEStateOnce, 2)
		return &PROBEStateValue
	}
	for PROBEatomic.LoadUint32(&PROBEStateOnce) != 2 {
		PROBEruntime.Gosched()
	}
	return &PROBEStateValue
}

func PROBEOpen() PROBEState {
	state := PROBEState{site: -1}
	path := PROBEos.Getenv("TMT_GO_TRACE")
	if path == "" {
		return state
	}
	f, err := PROBEos.OpenFile(path, PROBEos.O_RDWR, 0)
	if err != nil {
		panic(err)
	}
	defer f.Close()
	info, err := f.Stat()
	if err != nil {
		panic(err)
	}
	data, err := PROBEsyscall.Mmap(int(f.Fd()), 0, int(info.Size()), PROBEsyscall.PROT_READ|PROBEsyscall.PROT_WRITE, PROBEsyscall.MAP_SHARED)
	if err != nil {
		panic(err)
	}
	state.slots = PROBEunsafe.Slice((*uint64)(PROBEunsafe.Pointer(&data[0])), len(data)/8)
	if mode := PROBEos.Getenv("TMT_GO_MUTATION"); mode != "" {
		fields := PROBEstrings.Split(mode, ":")
		if len(fields) != 4 {
			panic("invalid mutation selector")
		}
		state.site, err = PROBEstrconv.Atoi(fields[0])
		if err != nil {
			panic(err)
		}
		state.mask, err = PROBEstrconv.ParseUint(fields[1], 10, 64)
		if err != nil {
			panic(err)
		}
		state.pivot, err = PROBEstrconv.ParseUint(fields[2], 10, 64)
		if err != nil {
			panic(err)
		}
		switch fields[3] {
		case "false", "0":
			state.mode = 0
		case "true", "1":
			state.mode = 1
		case "margin-down":
			state.mode = 2
		case "margin-up":
			state.mode = 3
		default:
			panic("invalid mutation selector")
		}
	}
	return state
}

func PROBERank[T PROBEInteger](value T) uint64 {
	var zero T
	if ^zero < zero {
		return uint64(value) ^ (1 << 63)
	}
	return uint64(value)
}

func PROBEOrder[T PROBEInteger](left, right T) uint64 {
	if left < right {
		return 1
	}
	if left > right {
		return 4
	}
	return 2
}

func PROBECompare[T PROBEInteger](site int, mask uint64, left, right T) uint64 {
	order := PROBEOrder(left, right)
	state := PROBEGetState()
	if state.slots == nil || site < 0 || site*5+5 > len(state.slots) {
		return order & mask
	}
	if state.site >= 0 {
		if state.site != site {
			return order & mask
		}
		mutantOrder := order
		switch state.mode {
		case 1:
			mutantOrder = PROBEOrder(PROBERank(left), state.pivot)
		case 2:
			lRank, rRank := PROBERank(left), PROBERank(right)
			if rRank == 0 {
				mutantOrder = 4
			} else {
				mutantOrder = PROBEOrder(lRank, rRank-1)
			}
		case 3:
			lRank, rRank := PROBERank(left), PROBERank(right)
			if rRank == ^uint64(0) {
				mutantOrder = 1
			} else {
				mutantOrder = PROBEOrder(lRank, rRank+1)
			}
		}
		return PROBEActivate(state, site, mask, order, mutantOrder)
	}
	slot := state.slots[site*5 : site*5+5]
	if order == 1 {
		lRank, rRank := PROBERank(left), PROBERank(right)
		PROBEMax(&slot[1], lRank)
		PROBEMax(&slot[3], ^(rRank - lRank))
	} else if order == 4 {
		lRank, rRank := PROBERank(left), PROBERank(right)
		PROBEMax(&slot[2], ^lRank)
		PROBEMax(&slot[4], ^(lRank - rRank))
	}
	PROBERecord(&slot[0], order)
	return order & mask
}

func PROBEMax(word *uint64, value uint64) {
	for old := PROBEatomic.LoadUint64(word); old < value; old = PROBEatomic.LoadUint64(word) {
		if PROBEatomic.CompareAndSwapUint64(word, old, value) {
			return
		}
	}
}

func PROBEActivate(state *PROBEState, site int, mask, order, mutantOrder uint64) uint64 {
	flags := uint64(8)
	mutant := mutantOrder & state.mask
	if (order&mask != 0) != (mutant != 0) {
		flags |= 16
	}
	PROBERecord(&state.slots[site*5], flags)
	return mutant
}

func PROBERecord(word *uint64, bits uint64) {
	for {
		old := PROBEatomic.LoadUint64(word)
		if old&bits == bits {
			return
		}
		if PROBEatomic.CompareAndSwapUint64(word, old, old|bits) {
			return
		}
	}
}
