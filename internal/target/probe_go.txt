package PROBE_PACKAGE

import (
	PROBEos "os"
	PROBEstrconv "strconv"
	PROBEstrings "strings"
	PROBEatomic "sync/atomic"
	PROBEsyscall "syscall"
	PROBEunsafe "unsafe"
)

type PROBEInteger interface {
	~int | ~int8 | ~int16 | ~int32 | ~int64 | ~uint | ~uint8 | ~uint16 | ~uint32 | ~uint64 | ~uintptr
}

type PROBEState struct {
	slots []uint64
	site int
	mask uint64
	pivot uint64
	shift bool
}

var PROBEStateValue = PROBEOpen()

func PROBEOpen() PROBEState {
	state := PROBEState{site: -1}
	path := PROBEos.Getenv("TMT_GO_TRACE")
	if path == "" {
		return state
	}
	f, err := PROBEos.OpenFile(path, PROBEos.O_RDWR, 0)
	if err != nil {
		panic(err)
	}
	defer f.Close()
	info, err := f.Stat()
	if err != nil {
		panic(err)
	}
	data, err := PROBEsyscall.Mmap(int(f.Fd()), 0, int(info.Size()), PROBEsyscall.PROT_READ|PROBEsyscall.PROT_WRITE, PROBEsyscall.MAP_SHARED)
	if err != nil {
		panic(err)
	}
	state.slots = PROBEunsafe.Slice((*uint64)(PROBEunsafe.Pointer(&data[0])), len(data)/8)
	if mode := PROBEos.Getenv("TMT_GO_MUTATION"); mode != "" {
		fields := PROBEstrings.Split(mode, ":")
		if len(fields) != 4 {
			panic("invalid mutation selector")
		}
		state.site, err = PROBEstrconv.Atoi(fields[0])
		if err != nil {
			panic(err)
		}
		state.mask, err = PROBEstrconv.ParseUint(fields[1], 10, 64)
		if err != nil {
			panic(err)
		}
		state.pivot, err = PROBEstrconv.ParseUint(fields[2], 10, 64)
		if err != nil {
			panic(err)
		}
		state.shift, err = PROBEstrconv.ParseBool(fields[3])
		if err != nil {
			panic(err)
		}
	}
	return state
}

func PROBERank[T PROBEInteger](value T) uint64 {
	var zero T
	if ^zero < zero {
		return uint64(value) ^ (1 << 63)
	}
	return uint64(value)
}

func PROBEOrder[T PROBEInteger](left, right T) uint64 {
	if left < right {
		return 1
	}
	if left > right {
		return 4
	}
	return 2
}

func PROBECompare[T PROBEInteger](site int, mask uint64, left, right T) uint64 {
	order := PROBEOrder(left, right)
	state := &PROBEStateValue
	if state.slots == nil {
		return order & mask
	}
	if state.site >= 0 {
		if state.site != site {
			return order & mask
		}
		mutantOrder := order
		if state.shift {
			mutantOrder = PROBEOrder(PROBERank(left), state.pivot)
		}
		return PROBEActivate(state, site, mask, order, mutantOrder)
	}
	slot := state.slots[site*3 : site*3+3]
	if order == 1 {
		value := PROBERank(left)
		for old := PROBEatomic.LoadUint64(&slot[1]); old < value; old = PROBEatomic.LoadUint64(&slot[1]) {
			if PROBEatomic.CompareAndSwapUint64(&slot[1], old, value) {
				break
			}
		}
	} else if order == 4 {
		value := ^PROBERank(left)
		for old := PROBEatomic.LoadUint64(&slot[2]); old < value; old = PROBEatomic.LoadUint64(&slot[2]) {
			if PROBEatomic.CompareAndSwapUint64(&slot[2], old, value) {
				break
			}
		}
	}
	PROBERecord(&slot[0], order)
	return order & mask
}

func PROBEActivate(state *PROBEState, site int, mask, order, mutantOrder uint64) uint64 {
	flags := uint64(8)
	mutant := mutantOrder & state.mask
	if (order&mask != 0) != (mutant != 0) {
		flags |= 16
	}
	PROBERecord(&state.slots[site*3], flags)
	return mutant
}

func PROBERecord(word *uint64, bits uint64) {
	if PROBEatomic.LoadUint64(word)&bits != bits {
		PROBEatomic.OrUint64(word, bits)
	}
}
